357  clear_if(ec);
358

360  throws_if(ec, Errors::CodecUnsupportedOperation);

380

382  // Output queue is full: drain it first, then call flush() again
383  return false;
384  }
385  if (ret == AVERROR_EOF) {
386  // Already drained, flush is idempotent
387  return true;
388  }
389  if (ret) {
