890 // Output queue is full: drain it first, then call flush() again
891 return false;
892 }
893 if (ret == AVERROR_EOF) {
894 // Already drained, flush is idempotent
895 return true;
896 }
897 if (ret) {
898 throws_if(ec, ret, ffmpeg_category());
899 return false;

1932 this->m_raw->subtitle_header_size = static_cast<int>(header.size());
