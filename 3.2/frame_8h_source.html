508class Frame : public FrameCommon

510protected:
518 T& assignOperator(const T &rhs) {
519 if (this == &rhs)
520 return static_cast<T&>(*this);
521 T(rhs).swap(static_cast<T&>(*this));
522 return static_cast<T&>(*this);
523 }

532 T& moveOperator(T &&rhs) {
533 if (this == &rhs)
534 return static_cast<T&>(*this);
535 T(std::move(rhs)).swap(static_cast<T&>(*this));
536 return static_cast<T&>(*this);
537 }
