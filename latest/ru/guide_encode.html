Кодирование

Кодирование берет кадры и выдает пакеты. Настройка длиннее, чем у декодирования, потому что выход надо описать: контейнер, кодек, разрешение, базу времени. Зато сам цикл короткий.

Настройка

OutputFormat ofrmt;
FormatContext octx;
ofrmt.setFormat(string(), out);
octx.setFormat(ofrmt);
Codec ocodec = findEncodingCodec(ofrmt);
VideoEncoderContext encoder {ocodec};
// Settings
encoder.setWidth(vdec.width());
encoder.setHeight(vdec.height());
if (vdec.pixelFormat() > -1)
encoder.setPixelFormat(vdec.pixelFormat());
encoder.setTimeBase(Rational{1, 1000});
encoder.setBitRate(vdec.bitRate());
encoder.open(Codec(), ec);
if (ec) {
cerr << "Can't opent encodec\n";
return 1;
}
Stream ost = octx.addStream(encoder);
ost.setFrameRate(vst.frameRate());
octx.openOutput(out, ec);
if (ec) {
cerr << "Can't open output\n";
return 1;
}
octx.dump();
octx.writeHeader();
octx.flush();

Участвуют три объекта:

  • av::OutputFormat задает контейнер. setFormat(string(), out) заполняет его по имени выходного файла; явное имя в конструкторе заставляет контейнер, который расширение не подразумевает.
  • av::FormatContext владеет выходом. setFormat() обязателен до addStream().
  • av::VideoEncoderContext создается из кодека, который вернул findEncodingCodec(). Некоторым кодекам нужно больше, чем параметры выше: аппаратные случаи и требование частоты кадров у QSV разобраны в Аппаратное кодирование и декодирование.

Порядок важен. Сначала open() кодировщика, потом addStream(), а openOutput() — уже когда потоки существуют, потому что муксер должен их видеть. writeHeader() начинает контейнер; dump() — диагностика, печатающая раскладку потоков, ее можно убрать.

Цикл

do {
// DECODING
auto frame = vdec.decode(pkt, ec);
count++;
//if (count > 200)
// break;
bool flushEncoder = false;
if (ec) {
cerr << "Decoding error: " << ec << endl;
return 1;
} else if (!frame) {
//cerr << "Empty frame\n";
//flushDecoder = false;
//continue;
if (flushDecoder) {
flushEncoder = true;
}
}
if (frame) {
clog << "Frame: pts=" << frame.pts() << " / " << frame.pts().seconds() << " / " << frame.timeBase() << ", " << frame.width() << "x" << frame.height() << ", size=" << frame.size() << ", ref=" << frame.isReferenced() << ":" << frame.refCount() << " / type: " << frame.pictureType() << endl;
// Change timebase
frame.setTimeBase(encoder.timeBase());
frame.setStreamIndex(0);
frame.setPictureType();
clog << "Frame: pts=" << frame.pts() << " / " << frame.pts().seconds() << " / " << frame.timeBase() << ", " << frame.width() << "x" << frame.height() << ", size=" << frame.size() << ", ref=" << frame.isReferenced() << ":" << frame.refCount() << " / type: " << frame.pictureType() << endl;
}
if (frame || flushEncoder) {
do {
// Encode
Packet opkt = frame ? encoder.encode(frame, ec) : encoder.encode(ec);
if (ec) {
cerr << "Encoding error: " << ec << endl;
return 1;
} else if (!opkt) {
//cerr << "Empty packet\n";
//continue;
break;
}
// Only one output stream
opkt.setStreamIndex(0);
clog << "Write packet: pts=" << opkt.pts() << ", dts=" << opkt.dts() << " / " << opkt.pts().seconds() << " / " << opkt.timeBase() << " / st: " << opkt.streamIndex() << endl;
octx.writePacket(opkt, ec);
if (ec) {
cerr << "Error write packet: " << ec << ", " << ec.message() << endl;
return 1;
}
} while (flushEncoder);
}
if (flushEncoder)
break;
} while (flushDecoder);

Места, где легко ошибиться:

  • База времени. Кодировщик настроен с базой времени, и кадр должен нести ту же, прежде чем попасть в кодировщик: frame.setTimeBase(encoder.timeBase()). Несовпадение дает пакеты с бессмысленными метками, а не ошибку.
  • Индекс потока. Packet::setStreamIndex() выбирает выходной поток. При одном потоке это 0.
  • Слив. encoder.encode() без кадра осушает кодировщик, и нужен do/while, потому что один вызов слива может вернуть несколько пакетов. В примере слив происходит в ветке «декодер в конце файла», где и появляются последние кадры.
  • Запись. octx.writePacket(pkt) тоже принимает код ошибки. Запись буферизуется; ничего не финально до writeTrailer().

Завершение

writeTrailer() пишет индекс контейнера и осушает муксер. Без него MP4-файл не проигрывается, хотя размер выглядит правдоподобно. Если муксер буферизует пакеты, writePacket() без аргументов осушает их до трейлера.

Аудио

av::AudioEncoderContext работает так же: задайте setSampleRate(), setChannelsCount(), setSampleFormat() и вызовите open(). Перед кодированием приведите входные сэмплы к раскладке кодировщика через av::AudioResampler, если они уже не совпадают.
