Фильтры

Граф фильтров стоит между декодером и кодировщиком: кадры входят через источник, проходят через любое число фильтров, описанных строкой, и выходят через сток.

Сборка графа

// Setup filter
Filter filter_buffer_src{"buffer"};
Filter filter_buffer_sink{"buffersink"};
FilterGraph filter_graph;
// Input and ouput contexts
std::stringstream src_args;
src_args << "video_size=" << vdec.width() << "x" << vdec.height() << ":pix_fmt=" << vdec.pixelFormat()
<< ":time_base=" << vdec.timeBase() << ":pixel_aspect=" << vdec.sampleAspectRatio();
clog << "Input arguments: " << src_args.str() << endl;
FilterContext src_ctx = filter_graph.createFilter(filter_buffer_src, "in", src_args.str(), ec);
if (ec)
{
clog << "Error creating filter source context: " << ec << ", " << ec.message() << endl;
return 1;
}
FilterContext sink_ctx = filter_graph.createFilter(filter_buffer_sink, "out", "", ec);
if (ec)
{
clog << "Error creating filter sink context: " << ec << ", " << ec.message() << endl;
return 1;
}
// Setup the filter chain
string filterChain;
if (argc > 3) {
filterChain = argv[3];
} else {
filterChain = "scale=320x200:-1:flags=lanczos,split [s0][s1]; "
"[s0] palettegen [p]; [s1][p] paletteuse[f]; ";
}
cout << "Filter: " << filterChain << endl;
filter_graph.parse(filterChain, src_ctx, sink_ctx, ec);
if (ec) {
clog << "Error parsing filter chain: " << ec << ", " << ec.message() << endl;
return 1;
}
filter_graph.config(ec);
if (ec)
{
clog << "Error configuring filter chain: " << ec << ", " << ec.message() << endl;
return 1;
}
// Setup the entry/exit points
BufferSrcFilterContext buffer_src{src_ctx};
BufferSinkFilterContext buffer_sink{sink_ctx};

Четыре шага:

  1. Назвать концы. Filter{"buffer"} и Filter{"buffersink"} — стандартные фильтры входа и выхода любого графа. В строку цепочки они не входят.
  2. Описать вход. Аргументы источника должны совпадать с кадрами, которые будут подаваться: размер, формат пикселей, база времени и соотношение сторон. Несовпадение проявится на config() или, хуже, даст молча неверный граф.
  3. Разобрать цепочку. parse(chain, source, sink) разрешает имена фильтров и связывает граф. Синтаксис цепочки — собственный синтаксис FFmpeg: scale=320x200:-1:flags=lanczos,split [s0][s1]; ....
  4. Сконфигурировать. config() согласует форматы и выделяет фильтры. До него подавать кадры нельзя.

BufferSrcFilterContext и BufferSinkFilterContext — типизированные дескрипторы двух концов. Стройте их после config(), из тех же контекстов.

Подача и чтение

// Push into the filter
cout << "Pushing frame into filter: " << frame.width() << "x" << frame.height() << ", size=" << frame.size()
<< ", ts=" << frame.pts() << ", tb: " << frame.timeBase() << endl;
buffer_src.addVideoFrame(frame, AV_BUFFERSRC_FLAG_KEEP_REF, ec);
if (ec)
{
clog << "Filter push error: " << ec << ", " << ec.message() << endl;
return 1;
}
// Pull until there are no more incoming frames
VideoFrame filtered;
while (buffer_sink.getVideoFrame(filtered, ec))
{
filtered.setTimeBase(encoder.timeBase());
auto ts = filtered.pts();
clog << " filtered: " << filtered.width() << "x" << filtered.height() << ", size=" << filtered.size()
<< ", ts=" << ts << ", tm: " << ts.seconds() << ", tb: " << filtered.timeBase()
<< ", ref=" << filtered.isReferenced() << ":" << filtered.refCount() << endl;
// Encode
Packet opkt = filtered ? encoder.encode(filtered, ec) : encoder.encode(ec);
if (ec) {
cerr << "Encoding error: " << ec << endl;
return 1;
}
if (opkt) {
// Only one output stream
opkt.setStreamIndex(0);
clog << "Write packet: pts=" << opkt.pts() << ", dts=" << opkt.dts() << " / " << opkt.pts().seconds() << " / " << opkt.timeBase() << " / st: " << opkt.streamIndex() << endl;
octx.writePacket(opkt, ec);
if (ec) {
cerr << "Error write packet: " << ec << ", " << ec.message() << endl;
return 1;
}
}
}

Один поданный кадр не означает один кадр на выходе: split, fps и фильтры с таймлайном меняют соотношение, а сток может вернуть несколько кадров после одной подачи. Внутренний цикл вызывает getVideoFrame(), пока тот не вернет false, и этот false — нормальный случай «больше выхода сейчас нет».

AV_BUFFERSRC_FLAG_KEEP_REF сохраняет поданный кадр валидным после вызова — это нужно примеру, потому что он дальше использует кадр декодера. Если кадр больше не нужен, флаг можно не ставить.

EAGAIN от стока — не ошибка: пример явно проверяет ec.value() != -EAGAIN, потому что сток сообщает его как значение, пока графу не хватает входа.

Временные метки

Отфильтрованные кадры несут собственные метки, полученные из базы времени входа. Если у кодировщика другая база времени, задайте ее на каждом отфильтрованном кадре перед кодированием — точно так же, как для обычных декодированных кадров.

Текст и аудио

Тот же граф работает для аудио с buffer/abuffersink, addAudioSamples() и getAudioSamples() (последний принимает количество сэмплов). Текстовые фильтры (например subtitles=) настраиваются строкой цепочки и цикла по кадрам не требуют.
